+);

+CREATE OR REPLACE FUNCTION public.sync_staff_shop_ids()
+ RETURNS SETOF uuid
+ LANGUAGE sql
+ STABLE SECURITY DEFINER
+ SET search_path TO 'public'
+AS $function$;
+  select sh.id
+  from public.staff s
+  join public.shops sh on sh.business_id = s.business_id and sh.active
+  where s.auth_user_id = auth.uid()
+    and s.active
+    and s.terminated_at is null
+    and (
+      s.role = 'owner'
+      or exists (
+        select 1 from public.staff_shops ss
+        where ss.staff_id = s.id and ss.shop_id = sh.id
+      )
+    )
+$function$;

+RETURNS trigger LANGUAGE plpgsql SET search_path = '' AS $$
+BEGIN
+  IF TG_OP = 'INSERT' OR public.sync_normalize_payload(NEW.payload)
+       IS DISTINCT FROM public.sync_normalize_payload(OLD.payload) THEN
+    NEW.server_seq := nextval('public.sync_records_server_seq_seq');
+  END IF;
+  RETURN NEW;
+END; $$;

+ RETURNS jsonb
+ LANGUAGE plpgsql
+ SECURITY DEFINER
+ SET search_path TO ''
+AS $function$
+declare
+  v_staff       record;
+  v_shop        uuid;
+  v_env         jsonb;
+  v_table       text;
+  v_uid         text;
+  v_payload     jsonb;
+  v_parked_seq  bigint;
+  v_existing    public.sync_records%rowtype;
+  v_applied     int := 0;
+  v_conflicts   int := 0;
+begin
+  select s.id, s.business_id, s.role, s.username into v_staff
+    from public.staff s
+   where s.auth_user_id = auth.uid() and s.active and s.terminated_at is null;
+  if not found then
+    return jsonb_build_object('ok', false, 'error', jsonb_build_object(
+      'code', 'not_authorized', 'message', 'No active staff profile'));
+  end if;
+
+  v_shop := shop_id;
+  if v_shop is null or v_shop not in (select public.sync_staff_shop_ids()) then
+    return jsonb_build_object('ok', false, 'error', jsonb_build_object(
+      'code', 'shop_required', 'message',
+      'The request must name a shop this account may write to'));
+  end if;
+
+  if jsonb_typeof(envelopes) <> 'array' or jsonb_array_length(envelopes) > 500 then
+    return jsonb_build_object('ok', false, 'error', jsonb_build_object(
+      'code', 'too_many_envelopes', 'message',
+      'Send between 0 and 500 envelopes per call'));
+  end if;
+
+  -- INVARIANT 1: seq order == commit order per shop
+  perform pg_advisory_xact_lock(hashtextextended(v_shop::text, 0));
+
+  for v_env in select * from jsonb_array_elements(envelopes) loop
+    v_table      := v_env->>'table';
+    v_uid        := v_env->>'uid';
+    v_payload    := v_env->'payload';
+    v_parked_seq := nullif(v_env->>'parked_seq', '')::bigint;
+
+    if v_table is null or v_uid is null
+       or jsonb_typeof(v_payload) <> 'object' then
+      return jsonb_build_object('ok', false, 'error', jsonb_build_object(
+        'code', 'malformed_envelope', 'message',
+        'Every envelope needs table, uid and a payload object'));
+    end if;
+
+    if v_table not in (

+      'deliveries', 'attendance_records', 'delivery_status_history',
+      'daily_close_actions', 'physical_cash_counts', 'settings', 'store_profile'
+    ) then
+      return jsonb_build_object('ok', false, 'error', jsonb_build_object(
+        'code', 'unknown_table', 'message', 'Unknown synced table: ' || v_table));
+    end if;
+
+    -- D8 (unchanged): corrections of recorded history from privileged sources only
+    if v_table = 'events'

+    end if;
+

+        return jsonb_build_object('ok', false, 'error', jsonb_build_object(

+CREATE OR REPLACE FUNCTION public.sync_pull(since_seq bigint DEFAULT 0, max_rows integer DEFAULT 200, shop uuid DEFAULT NULL::uuid)
+ RETURNS jsonb
+ LANGUAGE sql
+ STABLE
+ SET search_path TO 'public'
+AS $function$

+    'server_time_epoch_ms', (extract(epoch from now()) * 1000)::bigint,
+    'records', coalesce((
+      select jsonb_agg(jsonb_build_object(
+        'server_seq', r.server_seq,
+        'shop_id', r.shop_id,
+        'table', r.table_name,
+        'uid', r.row_uid,
+        'payload', r.payload,
+        'origin_device', r.origin_device,
+        'received_at', r.received_at
+      ) order by r.server_seq)
+      from (
+        select sr.* from public.sync_records sr
+        where sr.server_seq > since_seq
+          and (shop is null or sr.shop_id = shop)
+          and sr.shop_id in (select public.sync_staff_shop_ids())
+        order by sr.server_seq
+        limit least(greatest(coalesce(max_rows, 200), 1), 500)
+      ) r
+    ), '[]'::jsonb)
+  )
+$function$;
